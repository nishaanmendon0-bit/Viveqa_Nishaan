module pri_encoder_8x3(D, Y, Valid);
    input [7:0] D;
    output reg [2:0] Y;  
    output reg Valid;    

    always @(*) begin
        Valid = 1'b1; 
        
        if      (D[7]) Y = 3'b111;
        else if (D[6]) Y = 3'b110;
        else if (D[5]) Y = 3'b101;
        else if (D[4]) Y = 3'b100;
        else if (D[3]) Y = 3'b011;
        else if (D[2]) Y = 3'b010;
        else if (D[1]) Y = 3'b001;
        else if (D[0]) Y = 3'b000;
        else begin
            Y = 3'b000;
            Valid = 1'b0; 
        end
    end

endmodule
