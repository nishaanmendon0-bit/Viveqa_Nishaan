module jk_ff(J, K, clk, rst, Q, Qbar);
    input J, K;
    input clk;
    input rst;  
    output reg Q;
    output Qbar;

    parameter HOLD   = 2'b00;
    parameter RESET  = 2'b01;
    parameter SET    = 2'b10;
    parameter TOGGLE = 2'b11;

    
    assign Qbar = ~Q;

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            Q <= 1'b0; 
        end else begin
            case ({J, K})
                HOLD:   Q <= Q;       
                RESET:  Q <= 1'b0;    
                SET:    Q <= 1'b1;    
                TOGGLE: Q <= ~Q;      
            endcase
        end
    end

endmodule
