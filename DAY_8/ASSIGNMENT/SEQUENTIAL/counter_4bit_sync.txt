module counter_4bit_sync_load(clk, rst, load, data_in, count);
    input clk;
    input rst;               
    input load;              
    input [3:0] data_in;     
    output reg [3:0] count;  

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            count <= 4'b0000;        
        end else if (load) begin
            count <= data_in;        
        end else begin
            count <= count + 1'b1;   
        end
    end

endmodule
